# Hyprland session, installed by system/02-desktop.sh. One package per line: why it's here.

# Compositor and session
hyprland                     # the compositor
xdg-desktop-portal-hyprland  # screen sharing, screenshots for apps; file pickers come from the GTK/KDE portals
hyprpolkitagent              # password prompts when an app needs admin rights
hyprlock                     # lock screen
hypridle                     # lock / screen off / sleep after idle
hyprshutdown                 # clean log out (Super+Shift+E)
hyprpaper                    # wallpaper
qt6-wayland                  # Qt apps (Quickshell, Dolphin) run natively on Wayland
base-devel                   # g++ and pkg-config to build plugins/ (title-bar minimise) via hypr-plugins

# Look
adw-gtk-theme                # GTK3 apps match libadwaita's look, dark and light
papirus-icon-theme           # icon theme for the shell, dock and GTK apps
inter-font                   # UI font
cantarell-fonts              # "Cantarell" choice under Appearance > Font
noto-fonts                   # "Noto Sans" choice under Appearance > Font; fallback for missing glyphs
ttf-jetbrains-mono-nerd      # terminal / code font with icons
ttf-firacode-nerd            # second code font (ghostty / editor choice); nothing in the repo needs it

# Shell: bar, dock, launcher, notifications, OSD, Settings app
quickshell                   # the whole shell (home/quickshell)
fuzzel                       # emoji picker window (emojipick)
pavucontrol                  # full sound mixer (floating window rule in rules.lua)
nm-connection-editor         # advanced network settings (VPN, static IP), from Settings > Network
squeekboard                  # on-screen keyboard for tablet mode (the X13 flips into a tablet)
bluez                        # Bluetooth stack; the shell's Bluetooth controls talk to it over D-Bus

# Apps the keybinds open
ghostty                      # terminal (Super+Return)
dolphin                      # file manager (Super+E)

# Screenshots, clipboard, colour picker, OCR
flameshot                    # region screenshots with annotation (Print, Alt+Shift+S)
grim                         # screen capture for "copy text from screen" (screentext)
slurp                        # pick the region for screentext
wl-clipboard                 # wl-copy / wl-paste, clipboard history (clipstore)
wtype                        # types picked emoji and clipboard entries into apps
hyprpicker                   # colour picker (Super+Shift+C)
tesseract                    # OCR for "copy text from screen" (Super+Shift+T)
tesseract-data-eng           # English OCR data

# Small tools the scripts and shell call
jq                           # JSON in scripts (hyprctl -j, settings.json)
curl                         # weather and holidays in the shell
libnotify                    # notify-send from scripts
libqalculate                 # calculator in Spotlight (qalc)
brightnessctl                # brightness keys
playerctl                    # media keys and the media card
hyprsunset                   # Night Light
stow                         # links home/ into ~ (install.sh)
ddcutil                      # monitor input switch over DDC/CI (monswitch, Super+M)
xdg-utils                    # xdg-open: Spotlight file results, Settings > About "Open folder"
glib2                        # gsettings: GTK light/dark, accent, icon theme (autostart, ThemeSync)
fastfetch                    # system summary in the terminal (home/fastfetch config)
